// FUNÇÃO DE INTEGRAÇÃO COM GEMINI API (CORRIGIDA) async function gerarAnaliseIA(combinacao) { const modal = document.getElementById('aiModal'); const modalTitle = document.getElementById('modalTitle'); const modalBody = document.getElementById('modalBody'); modalTitle.textContent = `✦ Análise: ${combinacao}`; modalBody.innerHTML = `

Consultando as estrelas com Gemini IA...

`; modal.classList.add('active'); const promptText = `Atue como um astrólogo experiente e analítico. Faça uma análise detalhada e perspicaz da seguinte posição astrológica: "${combinacao}". Siga estas diretrizes de resposta: - Escreva de 2 a 3 parágrafos curtos. - Explique o significado psicológico, os desafios e os pontos fortes dessa combinação. - Use formatação HTML básica com parágrafos

e destaques em para facilitar a leitura. - Mantenha um tom profissional, acolhedor e profundo.`; // Garante que a chave limpa seja codificada para a URL const apiKey = GEMINI_API_KEY.trim(); const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${encodeURIComponent(apiKey)}`; try { const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [ { parts: [{ text: promptText }] } ] }) }); const data = await response.json(); if (!response.ok) { // Exibe o erro exato retornado pelo servidor do Google const errorMsg = data.error?.message || "Erro desconhecido na requisição."; modalBody.innerHTML = `

Erro na API (${response.status}): ${errorMsg}

`; return; } if (data.candidates && data.candidates[0]?.content?.parts[0]?.text) { let textResult = data.candidates[0].content.parts[0].text; textResult = textResult.replace(/```html/g, '').replace(/```/g, ''); modalBody.innerHTML = textResult; } else { modalBody.innerHTML = `

Resposta vazia da IA.

`; } } catch (err) { console.error(err); modalBody.innerHTML = `

Erro ao conectar com a API: ${err.message}

`; } }
